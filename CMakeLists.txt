cmake_minimum_required(VERSION 3.16)
project(advanced_algorithms LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs videoio)
find_package(Eigen3 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED gstreamer-app-1.0 gstreamer-video-1.0)
pkg_check_modules(DRM REQUIRED libdrm)
find_package(Threads REQUIRED)
add_library(bytetrack vendor/bytetrack/src/BYTETracker.cpp vendor/bytetrack/src/STrack.cpp vendor/bytetrack/src/kalmanFilter.cpp vendor/bytetrack/src/lapjv.cpp vendor/bytetrack/src/utils.cpp)
target_include_directories(bytetrack PUBLIC vendor/bytetrack/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(bytetrack PUBLIC Eigen3::Eigen ${OpenCV_LIBS})
add_executable(advanced_vision common/main.cpp common/camera.cpp common/preprocess.cpp common/model.cpp common/display.cpp)
target_include_directories(advanced_vision PRIVATE common runtime/include runtime/include/rga ${GST_INCLUDE_DIRS} ${DRM_INCLUDE_DIRS})
target_compile_options(advanced_vision PRIVATE -O2 -Wall -Wextra)
target_link_libraries(advanced_vision PRIVATE bytetrack Threads::Threads ${GST_LIBRARIES} ${DRM_LIBRARIES} ${CMAKE_SOURCE_DIR}/runtime/lib/librknnrt.so ${CMAKE_SOURCE_DIR}/runtime/lib/librga.so)
set_target_properties(advanced_vision PROPERTIES BUILD_RPATH "${CMAKE_SOURCE_DIR}/runtime/lib")
enable_testing()
add_executable(async_writer_tests common/async_writer_tests.cpp)
target_include_directories(async_writer_tests PRIVATE common)
target_link_libraries(async_writer_tests PRIVATE Threads::Threads)
add_test(NAME async_writer_contract COMMAND async_writer_tests)
set_tests_properties(async_writer_contract PROPERTIES TIMEOUT 5)
add_executable(npu_profile common/npu_profile.cpp)
target_include_directories(npu_profile PRIVATE common runtime/include ${OpenCV_INCLUDE_DIRS})
target_compile_options(npu_profile PRIVATE -O2 -Wall -Wextra)
target_link_libraries(npu_profile PRIVATE ${OpenCV_LIBS} ${CMAKE_SOURCE_DIR}/runtime/lib/librknnrt.so)
set_target_properties(npu_profile PROPERTIES BUILD_RPATH "${CMAKE_SOURCE_DIR}/runtime/lib")
add_executable(core_tests common/tests.cpp)
target_include_directories(core_tests PRIVATE common)
target_link_libraries(core_tests PRIVATE bytetrack Threads::Threads)
add_test(NAME core_contracts COMMAND core_tests)
add_executable(hardware_tests common/hardware_tests.cpp common/preprocess.cpp)
target_include_directories(hardware_tests PRIVATE common runtime/include/rga ${OpenCV_INCLUDE_DIRS})
target_link_libraries(hardware_tests PRIVATE ${OpenCV_LIBS} ${CMAKE_SOURCE_DIR}/runtime/lib/librga.so)
set_target_properties(hardware_tests PROPERTIES BUILD_RPATH "${CMAKE_SOURCE_DIR}/runtime/lib")
add_test(NAME rga_hardware COMMAND hardware_tests)
add_executable(render_tests common/render_tests.cpp)
target_include_directories(render_tests PRIVATE common ${OpenCV_INCLUDE_DIRS})
target_compile_options(render_tests PRIVATE -O2)
target_link_libraries(render_tests PRIVATE ${OpenCV_LIBS})
add_test(NAME vector_render_parity COMMAND render_tests)
foreach(app model_batch sr_tiled sr_tail_check)
 if(app STREQUAL "model_batch")
  set(app_source common/batch.cpp)
 elseif(app STREQUAL "sr_tail_check")
  set(app_source common/sr_tail_check.cpp)
 else()
  set(app_source common/sr_tiled.cpp)
 endif()
 add_executable(${app} ${app_source} common/model.cpp)
 target_include_directories(${app} PRIVATE common runtime/include ${OpenCV_INCLUDE_DIRS})
 target_link_libraries(${app} PRIVATE ${OpenCV_LIBS} Threads::Threads ${CMAKE_SOURCE_DIR}/runtime/lib/librknnrt.so)
 set_target_properties(${app} PROPERTIES BUILD_RPATH "${CMAKE_SOURCE_DIR}/runtime/lib")
endforeach()
